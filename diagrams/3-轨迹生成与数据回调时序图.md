sequenceDiagram
    participant M as SimManager
    participant G as TrajGenerator
    participant R as Ruckig OTG

    M->>G: generateTrajectory(params)
    G->>G: Validate and select DOF

    G->>R: Build InputParameter
    loop Each cycle
        R->>R: update(input, output)
        alt Calc
            R-->>G: Collect sample point
        else Final
            R-->>G: Complete
        end
    end

    G->>G: Resample trajectory
    G-->>M: Result(points, duration)

    loop Playback
        M->>M: getCurrentState(time)
    end
